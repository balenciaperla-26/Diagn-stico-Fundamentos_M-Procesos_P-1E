document.addEventListener('visibilitychange', () => { // Si hay un aviso de selección visible, NO contar como cambio const avisoSeleccion = document.querySelector('.aviso-seleccion'); if (document.hidden && E.pantalla === 'cuestionario' && !avisoSeleccion) { E.cambios++; mostrarAvisoCambioPantalla(); } }); document.addEventListener('fullscreenchange', () => { const avisoSeleccion = document.querySelector('.aviso-seleccion'); if (!document.fullscreenElement && E.pantalla === 'cuestionario' && !avisoSeleccion) { E.cambios++; mostrarAvisoCambioPantalla(); } });
